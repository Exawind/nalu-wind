24template <typename range_type, typename helper_type>
25void
26scale_lift_force(
27 ActuatorBulk& actBulk,
28 const ActuatorMeta& actMeta,
29 range_type& rangePolicy,
30 helper_type& helper,
31 const int offset,
32 const int nPoints)
33{
34 // suppress compiler warnings for unused variables when compiling w/o openfast
35 (void)offset;
36 (void)nPoints;
37
38 switch (actMeta.actuatorType_) {
39 case (ActuatorType::ActLineSimpleNGP): {
40 auto actBulkSimple = dynamic_cast<ActuatorBulkSimple&>(actBulk);
41 auto actMetaSimple = dynamic_cast<const ActuatorMetaSimple&>(actMeta);
42 auto G = helper.get_local_view(actBulkSimple.liftForceDistribution_);
43 auto rho = helper.get_local_view(actBulkSimple.density_);
44
45 const int turbId = actBulkSimple.localTurbineId_;
46 double dR = actMetaSimple.dR_.view_host()(turbId);
47
48 Kokkos::parallel_for("scale G", rangePolicy, [=](int i) {
49 const double denom = rho(i) * dR;
50 for (int j = 0; j < 3; ++j) {
51 G(i, j) /= denom;
52 }
53 });
54 break;
55 }
56 case (ActuatorType::ActLineFASTNGP):
57 case (ActuatorType::ActDiskFASTNGP): {
58#ifndef KYNEMA_UGF_USES_OPENFAST
59 STK_ThrowErrorMsg("Actuator methods require OpenFAST");
60#if !defined(KOKKOS_ENABLE_GPU)
61 break;
62#endif
63#else
64 auto G = helper.get_local_view(actBulk.liftForceDistribution_);
65 auto point = helper.get_local_view(actBulk.pointCentroid_);
66 Kokkos::parallel_for("scale G FAST outputs", rangePolicy, [=](int i) {
67 double dr = 0;
68 if (i == offset) {
69 for (int j = 0; j < 3; ++j) {
70 dr += std::pow(point(i, j) - point(i + 1, j), 2.0);
71 }
72 } else if (i == offset + nPoints - 1) {
73 for (int j = 0; j < 3; ++j) {
74 dr += std::pow(point(i, j) - point(i - 1, j), 2.0);
75 }
76 } else {
77 for (int j = 0; j < 3; ++j) {
78 dr += std::pow(point(i - 1, j) - point(i + 1, j), 2.0);
79 }
80 }
81 // dr is computed using central difference
82 dr = 0.5 * std::sqrt(dr);
83 for (int j = 0; j < 3; ++j) {
84 G(i, j) /= dr;
85 }
86 });
87 break;
88#endif
89 }
90 default: {
91 throw std::runtime_error("Unsupported actuator type supplied to the fllc");
92 }
93 }
94}
