49 template <typename LambdaFunction>
50 void run_algorithm(stk::mesh::BulkData& bulk_data, LambdaFunction lambdaFunc)
51 {
52 stk::mesh::MetaData& meta_data = bulk_data.mesh_meta_data();
53 const int nDim = meta_data.spatial_dimension();
54 const int lhsSize = rhsSize_ * rhsSize_;
55 const int scratchIdsSize = rhsSize_;
56
57 const stk::mesh::NgpMesh& ngpMesh = realm_.ngp_mesh();
58 const kynema_ugf_ngp::FieldManager& fieldMgr = realm_.ngp_field_manager();
59 ElemDataRequestsGPU dataNeededNGP(fieldMgr, dataNeededByKernels_);
60
61 const auto reqType = (entityRank_ == stk::topology::ELEM_RANK)
62 ? ElemReqType::ELEM
63 : ElemReqType::FACE;
64
65 const int bytes_per_team = 0;
66 const int bytes_per_thread = calculate_shared_mem_bytes_per_thread(
67 lhsSize, rhsSize_, scratchIdsSize, meta_data.spatial_dimension(),
68 dataNeededNGP, reqType);
69
70 stk::mesh::Selector elemSelector = meta_data.locally_owned_part() &
71 stk::mesh::selectUnion(partVec_) &
72 !realm_.get_inactive_selector();
73
74 const auto& elem_buckets =
75 stk::mesh::get_bucket_ids(bulk_data, entityRank_, elemSelector);
76
77 // Create local copies of class data
78 const auto entityRank = entityRank_;
79 const auto nodesPerEntity = nodesPerEntity_;
80 const auto rhsSize = rhsSize_;
81
82 auto team_exec = sierra::kynema_ugf::get_device_team_policy(
83 elem_buckets.size(), bytes_per_team, bytes_per_thread);
84 Kokkos::parallel_for(
85 team_exec,
86 KOKKOS_LAMBDA(const sierra::kynema_ugf::DeviceTeamHandleType& team) {
87 auto bktId = elem_buckets.device_get(team.league_rank());
88 auto& b = ngpMesh.get_bucket(entityRank, bktId);
89
90#if !defined(KOKKOS_ENABLE_GPU)
91 STK_ThrowAssertMsg(
92 b.topology().num_nodes() == (unsigned)nodesPerEntity,
93 "AssembleElemSolverAlgorithm expected nodesPerEntity_ = "
94 << nodesPerEntity
95 << ", but b.topology().num_nodes() = " << b.topology().num_nodes());
96#endif
97
98 SharedMemData<DeviceTeamHandleType, DeviceShmem> smdata(
99 team, nDim, dataNeededNGP, nodesPerEntity, rhsSize);
100
101 const size_t bucketLen = b.size();
102 const size_t simdBucketLen = get_num_simd_groups(bucketLen);
103
104 Kokkos::parallel_for(
105 Kokkos::TeamThreadRange(team, simdBucketLen),
106 [&](const size_t& bktIndex) {
107 int numSimdElems =
108 get_length_of_next_simd_group(bktIndex, bucketLen);
109 smdata.numSimdElems = numSimdElems;
110
111 for (int simdElemIndex = 0; simdElemIndex < numSimdElems;
112 ++simdElemIndex) {
113 stk::mesh::Entity element = b[bktIndex * simdLen + simdElemIndex];
114 const auto elemIndex = ngpMesh.fast_mesh_index(element);
115 smdata.ngpElemNodes[simdElemIndex] =
116 ngpMesh.get_nodes(entityRank, elemIndex);
117 fill_pre_req_data(
118 dataNeededNGP, ngpMesh, entityRank, element,
119 *smdata.prereqData[simdElemIndex]);
120 }
121
122#if !defined(KOKKOS_ENABLE_GPU)
123 // No need to interleave on GPUs
124 copy_and_interleave(
125 smdata.prereqData, numSimdElems, smdata.simdPrereqData);
126#endif
127
128 fill_master_element_views(dataNeededNGP, smdata.simdPrereqData);
129 lambdaFunc(smdata);
130 });
131 });
132 }

135 stk::mesh::EntityRank entityRank_;
136
138 double diagRelaxFactor_{1.0};
139 unsigned nodesPerEntity_;
140 int rhsSize_;
141};
