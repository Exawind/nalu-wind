46 template <typename LambdaFunction>
47 void
48 run_face_elem_algorithm(stk::mesh::BulkData& bulk, LambdaFunction lamdbaFunc)
49 {
50 int nDim = bulk.mesh_meta_data().spatial_dimension();
51
52 // Register face ME instance in elemdata also to obtain face integration
53 // points
54 if (elemDataNeeded_.get_cvfem_face_me() == nullptr)
55 elemDataNeeded_.add_cvfem_face_me(faceDataNeeded_.get_cvfem_face_me());
56
57 int rhsSize = nodesPerElem_ * numDof_, lhsSize = rhsSize * rhsSize,
58 scratchIdsSize = rhsSize;
59
60 const stk::mesh::NgpMesh& ngpMesh = realm_.ngp_mesh();
61 const kynema_ugf_ngp::FieldManager& fieldMgr = realm_.ngp_field_manager();
62 ElemDataRequestsGPU faceDataNGP(fieldMgr, faceDataNeeded_);
63 ElemDataRequestsGPU elemDataNGP(fieldMgr, elemDataNeeded_);
64
65 const int bytes_per_team = 0;
66 const int bytes_per_thread = calculate_shared_mem_bytes_per_thread(
67 lhsSize, rhsSize, scratchIdsSize, nDim, faceDataNGP, elemDataNGP);
68
69 const auto nodesPerFace = nodesPerFace_;
70 const auto nodesPerElem = nodesPerElem_;
71 stk::mesh::Selector s_locally_owned_union =
72 bulk.mesh_meta_data().locally_owned_part() &
73 stk::mesh::selectUnion(partVec_);
74 stk::mesh::EntityRank sideRank = bulk.mesh_meta_data().side_rank();
75 const auto& buckets =
76 stk::mesh::get_bucket_ids(bulk, sideRank, s_locally_owned_union);
77
78 auto team_exec = sierra::kynema_ugf::get_device_team_policy(
79 buckets.size(), bytes_per_team, bytes_per_thread);
80 Kokkos::parallel_for(
81 team_exec,
82 KOKKOS_LAMBDA(const sierra::kynema_ugf::DeviceTeamHandleType& team) {
83 auto bktId = buckets.device_get(team.league_rank());
84 auto& b = ngpMesh.get_bucket(sideRank, bktId);
85
86#if !defined(KOKKOS_ENABLE_GPU)
87 STK_ThrowAssertMsg(
88 b.topology().num_nodes() == (unsigned)nodesPerFace,
89 "AssembleFaceElemSolverAlgorithm expected nodesPerEntity = "
90 << nodesPerFace
91 << ", but b.topology().num_nodes() = " << b.topology().num_nodes());
92#endif
93
94 SharedMemData_FaceElem<DeviceTeamHandleType, DeviceShmem> smdata(
95 team, nDim, faceDataNGP, elemDataNGP, nodesPerFace, nodesPerElem,
96 rhsSize);
97
98 const size_t bucketLen = b.size();
99 const size_t simdBucketLen =
100 sierra::kynema_ugf::get_num_simd_groups(bucketLen);
101
102 Kokkos::parallel_for(
103 Kokkos::TeamThreadRange(team, simdBucketLen),
104 [&](const size_t& bktIndex) {
105 size_t simdGroupLen =
106 sierra::kynema_ugf::get_length_of_next_simd_group(
107 bktIndex, bucketLen);
108 size_t numFacesProcessed = 0;
109 do {
110 int elemFaceOrdinal = -1;
111 int simdFaceIndex = 0;
112 while ((numFacesProcessed + simdFaceIndex) < simdGroupLen) {
113 stk::mesh::Entity face =
114 b[bktIndex * simdLen + numFacesProcessed + simdFaceIndex];
115 const auto ngpFaceIndex = ngpMesh.fast_mesh_index(face);
116 // STK_ThrowAssertMsg(
117 // bulk.num_elements(face) == 1,
118 // "Expecting just 1 element attached to face!");
119 int thisElemFaceOrdinal =
120 ngpMesh.get_element_ordinals(sideRank, ngpFaceIndex)[0];
121
122 if (
123 elemFaceOrdinal >= 0 &&
124 thisElemFaceOrdinal != elemFaceOrdinal) {
125 break;
126 }
127
128 const auto& elems =
129 ngpMesh.get_elements(sideRank, ngpFaceIndex);
130 const auto elemIndex = ngpMesh.fast_mesh_index(elems[0]);
131
132 smdata.ngpConnectedNodes[simdFaceIndex] =
133 ngpMesh.get_nodes(stk::topology::ELEMENT_RANK, elemIndex);
134 smdata.elemFaceOrdinal = thisElemFaceOrdinal;
135 elemFaceOrdinal = thisElemFaceOrdinal;
136 sierra::kynema_ugf::fill_pre_req_data(
137 faceDataNGP, ngpMesh, sideRank, face,
138 *smdata.faceViews[simdFaceIndex]);
139 sierra::kynema_ugf::fill_pre_req_data(
140 elemDataNGP, ngpMesh, stk::topology::ELEMENT_RANK, elems[0],
141 *smdata.elemViews[simdFaceIndex]);
142 ++simdFaceIndex;
143 }
144 smdata.numSimdFaces = simdFaceIndex;
145 numFacesProcessed += simdFaceIndex;
146
147#if !defined(KOKKOS_ENABLE_GPU)
148 // No need to interleave on GPUs
149 copy_and_interleave(
150 smdata.faceViews, smdata.numSimdFaces, smdata.simdFaceViews);
151 copy_and_interleave(
152 smdata.elemViews, smdata.numSimdFaces, smdata.simdElemViews);
153#endif
154
155 fill_master_element_views(
156 faceDataNGP, smdata.simdFaceViews, smdata.elemFaceOrdinal);
157 fill_master_element_views(
158 elemDataNGP, smdata.simdElemViews, smdata.elemFaceOrdinal);
159
160 lamdbaFunc(smdata);
161 } while (numFacesProcessed < simdGroupLen);
162 });
163 });
164 }
165
166 ElemDataRequests faceDataNeeded_;
167 ElemDataRequests elemDataNeeded_;
168 double diagRelaxFactor_{1.0};
169 unsigned numDof_;
170 unsigned nodesPerFace_;
171 unsigned nodesPerElem_;
172 int rhsSize_;
173};
